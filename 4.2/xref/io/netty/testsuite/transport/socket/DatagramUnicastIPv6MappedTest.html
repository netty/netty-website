16  package io.netty.testsuite.transport.socket;
17  
18  import io.netty.channel.socket.DatagramChannel;
19  import io.netty.channel.socket.nio.NioDatagramChannel;
20  import io.netty.util.NetUtil;
21  import io.netty.util.internal.PlatformDependent;
22  
23  import java.io.IOException;
24  import java.io.UncheckedIOException;
25  import java.net.BindException;
26  import java.net.InetAddress;
27  import java.net.InetSocketAddress;
28  import java.net.SocketAddress;
29  import java.net.StandardProtocolFamily;
30  import java.net.UnknownHostException;
31  
32  public class DatagramUnicastIPv6MappedTest extends DatagramUnicastIPv6Test {
33  
34      private static final InetAddress IPV4_ANY;
35      private static final int MAX_PORT_SELECTION_ATTEMPTS = 10;
36  
37      static {
38          try {
39              IPV4_ANY = InetAddress.getByAddress(new byte[] {0, 0, 0, 0});
40          } catch (UnknownHostException e) {
41              throw new ExceptionInInitializerError(e);
42          }
43      }
44  
45      @Override
46      protected SocketAddress newSocketAddress() {
47          // Explicit use IPv4 wildcard for bind while the channel is using SocketProtocolFamily.INET6.
48          //
49          // Don't use port 0: when an IPv6 socket binds to the wildcard address, macOS only checks the ports of other
50          // IPv6 sockets when it selects the ephemeral port, so it might select a port that an IPv4 socket is
51          // bound to. Datagrams that are sent to the IPv4 loopback address are then delivered to the IPv4 socket and
52          // never reach the channel, no matter how often they are resent:
53          //
54          // - in6_pcbsetport() checks if the port is used with in6_pcblookup_local()...
55          //   https://github.com/apple-oss-distributions/xnu/blob/xnu-12377.121.6/bsd/netinet6/in6_src.c#L1779-L1780
56          // - ...which skips all sockets that are not IPv6 sockets.
57          //   https://github.com/apple-oss-distributions/xnu/blob/xnu-12377.121.6/bsd/netinet6/in6_pcb.c#L1297-L1301
58          // - The netns_reserve_in() check that follows only fails if Skywalk reserved the port, not BSD sockets.
59          //   https://github.com/apple-oss-distributions/xnu/blob/xnu-12377.121.6/bsd/skywalk/namespace/netns.c#L774-L783
60          // - When an IPv4 datagram is received, in_pcblookup_hash_locked() prefers IPv4 sockets bound to the
61          //   destination or the wildcard address over IPv6 sockets bound to the wildcard address.
62          //   https://github.com/apple-oss-distributions/xnu/blob/xnu-12377.121.6/bsd/netinet/in_pcb.c#L2634-L2666
63          return new InetSocketAddress(IPV4_ANY, unusedPort());
64      }
65  
66      /**
67       * Returns a port that is neither used by an IPv4 nor by an IPv6 socket.
68       */
69      private static int unusedPort() {
70          BindException cause = null;
71          for (int i = 0; i < MAX_PORT_SELECTION_ATTEMPTS; i++) {
72              int port;
73              // The IPv4 stack selects a port that is unused by IPv4 sockets and by IPv6 sockets that accept IPv4
74              // datagrams.
75              try (java.nio.channels.DatagramChannel ipv4 =
76                           java.nio.channels.DatagramChannel.open(StandardProtocolFamily.INET)) {
77                  ipv4.bind(new InetSocketAddress(IPV4_ANY, 0));
78                  port = ((InetSocketAddress) ipv4.getLocalAddress()).getPort();
79              } catch (IOException e) {
80                  throw new UncheckedIOException(e);
81              }
82              // Bind to the port like the channel will do, which fails if an IPv6 socket that only accepts IPv6
83              // datagrams uses it. The IPv4 channel must be closed by now, as this binds to the IPv4 wildcard address
84              // as well on some platforms (like Linux).
85              try (java.nio.channels.DatagramChannel ipv6 =
86                           java.nio.channels.DatagramChannel.open(StandardProtocolFamily.INET6)) {
87                  ipv6.bind(new InetSocketAddress(IPV4_ANY, port));
88                  return port;
89              } catch (BindException e) {
90                  cause = e;
91              } catch (IOException e) {
92                  throw new UncheckedIOException(e);
93              }
94          }
95          throw new UncheckedIOException("Failed to find a port that is unused by IPv4 and IPv6 sockets", cause);
96      }
97  
98      @Override
99      protected InetSocketAddress convertAnyAddress(InetSocketAddress serverAddress) {
100         InetAddress addr = serverAddress.getAddress();
101         if (addr.isAnyLocalAddress()) {
102             return new InetSocketAddress(NetUtil.LOCALHOST4, serverAddress.getPort());
103         }
104         return serverAddress;
105     }
106 
107     @Override
108     protected boolean disconnectMightFail(DatagramChannel channel) {
109         // See https://bugs.openjdk.org/browse/JDK-8285515
110         if (channel instanceof NioDatagramChannel && PlatformDependent.javaVersion() < 20) {
111             return true;
112         }
113         return super.disconnectMightFail(channel);
114     }
115 }
